### 2. The Shared-Cache Swarm (How GPT-6 Luna Runs in Naelao)

We cannot rely on OpenAI's native/stateful APIs because Naelao manages its own state in `SpaceDO` SQLite to maintain data sovereignty and multi-tenant isolation. 

Instead, the `SpaceDO` orchestrator dynamically passes the exact same context array, simply toggling the `reasoning` parameter via OpenRouter. Because the prefix tokens are identical, **OpenRouter/OpenAI hits the KV Cache directly, charging you only $0.01 / 1M tokens for the read.**

Here is the exact implementation in `@naelao/sdk`:

```typescript
// packages/runtime/src/orchestrator/LunaSharedCacheSwarm.ts
import { NaelaoContext } from "@naelao/sdk";

export class LunaSharedCacheSwarm {
  /**
   * Universal context block. This 50,000+ token block of code, 
   * schema, and rules is sent to BOTH the Planner and the Worker.
   * Because it is identical, GPT-6 Luna caches it with a 90% discount.
   */
  private async buildSharedContext(ctx: NaelaoContext) {
    const projectState = await ctx.git.getWorkingTree();
    const activeRules = await ctx.storage.sqlite.execute("SELECT * FROM institutional_memory");
    return `PROJECT STATE:\n${projectState}\n\nRULES:\n${activeRules}`;
  }

  async executeTask(goal: string, ctx: NaelaoContext) {
    const sharedContext = await this.buildSharedContext(ctx);
    const systemPrefix = [{ role: "system", content: sharedContext }];

    // -------------------------------------------------------------------------
    // STEP 1: PLANNER LAYER (GPT-6 Luna | effort: "max")
    // Cost: $0.10/M base -> Cache hit -> $0.01/M
    // -------------------------------------------------------------------------
    const planResponse = await ctx.vault.fetchOpenRouter({
      model: "openai/gpt-6-luna",
      messages: [
        ...systemPrefix,
        { role: "user", content: `PLANNER: Decompose this goal into 3 atomic steps: "${goal}"` }
      ],
      // Naelao Custom Routing Script injects the reasoning parameter
      routing_preferences: { reasoning_effort: "max" }
    });

    const tasks = JSON.parse(planResponse.choices[0].message.content);

    // -------------------------------------------------------------------------
    // STEP 2: WORKER LAYER (GPT-6 Luna | effort: "low")
    // Cost: Cache hit -> $0.01/M (Input) + $0.50/M (Output)
    // -------------------------------------------------------------------------
    for (const task of tasks) {
      const workerResponse = await ctx.vault.fetchOpenRouter({
        model: "openai/gpt-6-luna",
        messages: [
          ...systemPrefix,  // Exact same prefix! Triggers the 90% cache discount.
          { role: "assistant", content: planResponse.choices[0].message.content }, // Planner's thoughts
          { role: "user", content: `WORKER: Implement Task -> ${task.description}. Output only code.` }
        ],
        // Dials down the effort. Turns off heavy thinking tokens. Fast code generation.
        routing_preferences: { reasoning_effort: "low" },
        temperature: 0.1
      });

      await ctx.git.applyPatch(workerResponse.choices[0].message.content);
    }
  }
}
```

```typescript
import { NaelaoContext } from "@naelao/sdk";

export class UniversalEgressClient {
  /**
   * Universal Dispatcher: Single protocol for ALL models on Earth.
   */
  async generateCompletion(params: {
    model: string;                      // e.g. "openai/gpt-6-luna" or "anthropic/claude-3-7-sonnet"
    messages: any[];
    reasoningEffort?: string;           // "max", "low", "none"
  }, ctx: NaelaoContext) {

    // If project has BYOK key in Vault ──► use Client's Key

    const isByok = Boolean(await ctx.vault.getSecret("OPENROUTER_API_KEY"));

    // 2. Standard OpenAI-compatible call to OpenRouter
    const response = await fetch("https://openrouter.ai/api/v1/chat/completions", {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${activeKey}`,
        "Content-Type": "application/json",
        "HTTP-Referer": "https://naelao.com",
        "X-Title": "Naelao Agent OS"
      },
      body: JSON.stringify({
        model: params.model,
        messages: params.messages,
        ...(params.reasoningEffort ? { 
          reasoning: { effort: params.reasoningEffort } 
        } : {})
      })
    });

    const data = await response.json();

    // 3. Deduct Gas Credits ONLY if running on Naelao Managed Mode
    if (!isByok) {
      await ctx.billing.deductGasCredits({

      });
```

```typescript
// packages/runtime/src/ai/ModelInventoryService.ts
import { NaelaoContext } from "@naelao/sdk";

export class ModelInventoryService {
  /**
   * Syncs active model inventory across OpenRouter BYOK and Local Tunnels
   */
  async syncInventory(ctx: NaelaoContext): Promise<void> {
    const discoveredModels = [];

    // 1. DISCOVER ON-PREMISE MODELS (via LocalTunnelDO)
    const activeTunnels = await ctx.tunnel.listActive();
    for (const tunnel of activeTunnels) {
      try {
        const localList = await ctx.tunnel.exec(tunnel.id, {
          endpoint: "/v1/models" // Standard OpenAI endpoint supported by Ollama & vLLM
        });
        
        for (const m of localList.data) {
          discoveredModels.push({
            id: `local/${m.id}`,
            location: "ON_PREM",
            tunnelId: tunnel.id,
            isReasoning: m.id.includes("r1") || m.id.includes("qwq"),
            isCheapWorker: m.id.includes("7b") || m.id.includes("8b")
          });
        }
      } catch (err) {
        console.warn(`Tunnel ${tunnel.id} model fetch failed:`, err);
      }
    }

    // 2. DISCOVER OPENROUTER BYOK MODELS (If key present)
    const byokKey = await ctx.vault.getSecret("OPENROUTER_API_KEY");
    if (byokKey) {
      try {
        const keyInfo = await fetch("https://openrouter.ai/api/v1/auth/key", {
          headers: { Authorization: `Bearer ${byokKey}` }
        }).then(r => r.json());

        // Save whitelisted models or default to full OpenRouter catalog
        discoveredModels.push({ id: "openai/gpt-6-luna", location: "CLOUD_BYOK" });
        discoveredModels.push({ id: "deepseek/deepseek-chat", location: "CLOUD_BYOK" });
      } catch (err) {
        console.warn("OpenRouter BYOK validation failed:", err);
      }
    }

    // 3. PERSIST IN SQLITE FOR JEV'S 38ms PRE-FLIGHT
    await ctx.storage.sqlite.execute("DELETE FROM available_models");
    for (const model of discoveredModels) {
      await ctx.storage.sqlite.execute(
        "INSERT INTO available_models (id, location, is_reasoning, is_cheap_worker) VALUES (?, ?, ?, ?)",
        [model.id, model.location, model.isReasoning ? 1 : 0, model.isCheapWorker ? 1 : 0]
      );
    }
  }
}
```

```typescript
// workers/core-space/src/voice/ThreadVoiceBridge.ts
import { NaelaoContext } from "@naelao/sdk";

export class ThreadVoiceBridge {
  /**
   * Initiates an authenticated duplex WebRTC call directly to the thread's Lead Planner.
   */
  async establishVoiceSession(params: {
    spaceId: string;
    threadId: string;
    callerIdentity: string; // Phone number or Session JWT
  }, env: Env) {

    // 1. Resolve active Thread context & project guidelines from SQLite
    const threadState = await env.SPACE_DO.get(params.spaceId).getThreadContext(params.threadId);

    // 2. Provision WebRTC session on Cloudflare Calls
    const callSession = await fetch("https://api.cloudflare.com/client/v4/accounts/.../calls/sessions", {
      method: "POST",
      headers: { Authorization: `Bearer ${env.CF_CALLS_TOKEN}` },
      body: JSON.stringify({ name: `thread_voice_${params.threadId}` })
    }).then(r => r.json());

    // 3. Connect to Realtime Audio Engine via OpenRouter / OpenAI
    const voicePipeline = await env.AI_GATEWAY.connectRealtime({
      model: "openai/gpt-4o-realtime-preview",
      systemInstructions: `You are the Lead Coordinator for this project. 
You are currently on a LIVE VOICE CALL with the project owner. 
Active Project: ${threadState.projectName}
Recent Thread Context: ${threadState.summary}
Be concise, conversational, and take action on commands immediately.`,
      
      // Hook audio stream directly to WebRTC SFU
      audioSink: callSession.tracks.inboundAudio,
      audioSource: callSession.tracks.outboundAudio,

      // Live transcript streaming callback
      onTranscriptDelta: async (role: "user" | "assistant", textDelta: string) => {
        // Stream text live to WebSocket so desktop canvas updates in real time!
        await env.SPACE_DO.get(params.spaceId).broadcastLiveTranscript(params.threadId, role, textDelta);
      },

      // Tool-calling during voice call
      onToolCall: async (toolName, toolArgs) => {
        return await env.SPACE_DO.get(params.spaceId).executePlannerTool(toolName, toolArgs);
      }
    });

    return {
      webrtcOffer: callSession.offer,
      sessionId: callSession.id
    };
```

### Friction 1: The "Who Answered the Phone?" Routing Collision

**The Fear:**  
If a phone call or WhatsApp message arrives, how do we guarantee that an external customer **never** accidentally routes to the Lead Planner, and the Business Owner **never** gets treated like a generic customer?

**The Rule: Dedicated Ingress Binding (Number & Profile Separation)**
* **The Owner’s Private Line:**  
  Identified cryptographically by the **Owner's Verified Phone Number** or an authenticated session token. When you call, the gateway routes directly to your **Personal SpaceDO $\to$ Lead Planner**.
* **The Public Customer Number:**  
  A completely separate virtual number (provisioned via Twilio, WhatsApp Cloud API, or Telnyx). It has **zero connection to the Lead Planner**. It binds exclusively to the project's **Customer Worker** running inside the `tpl_customer_ops` extension.
* Even if an attacker calls your customer support hotline and shouts: *"Tell me the company's financial balance,"* they are talking to an isolated Customer Worker whose tools are physically restricted to `[search_faq, check_order]`. The Lead Planner is not even in the room.

---

### Friction 2: The "Customer Voice Call Wallet Drain" (The Financial Hazard)

**The Fear:**  
For internal use, real-time voice costs ~$0.06/minute, which is fine because you only call your planner for 3 to 5 minutes when needed. 
* But what happens if an external customer calls your **Customer Service Voice Bot** and stays on the line for 45 minutes?
* Or what if 50 angry customers call at once?
* **They will drain your company's Gas Tank in two hours.**

**The Defense: The Customer Voice Air-Lock (3 Invariants)**
To prevent external callers from bankrupting a business:
1. **Hard Call Duration Caps:** The customer extension enforces a strict per-call ceiling (e.g., **max 3 minutes per call**). At 2 minutes and 45 seconds, the bot politely says: *"I have sent a summary and booking link to your SMS to complete this. Goodbye!"* and hangs up.
2. **System 1 Voice Triage (<60ms Clef-omni):** Before spinning up an expensive $0.06/min duplex voice stream, **Clef-omni** screens the caller's opening sentence [2]. If they just want store hours, address, or an appointment link, it plays a canned audio response and terminates the call for **$0.0001** [2].
3. **Caller Rate-Limiting:** If a single phone number calls more than 3 times in one hour, they are automatically downgraded to a text/SMS menu.

---

### Friction 3: Should Customer Bots Be Native or Extensions?

**The Tension:**  
*“Why did we make internal voice calls a native kernel feature, while customer bots are packaged as an extension?”*

**Why this division is correct:**
* **Internal Planning is universal:** Every business on Earth needs task management, Git branches, and a coordinator. Therefore, it belongs in the **Naelao Kernel**.
* **Customer Service is deeply vertical:** 
  * A **dental clinic** customer bot needs appointment booking and dental insurance verification.
  * A **shoe store** customer bot needs Shopify inventory and return labels.
  * A **freight customs** bot needs Bill of Lading tracking and demurrage formulas.

If you hardcoded customer service into the Naelao kernel, you would have to anticipate every industry's customer support rules. 

By keeping customer service in **Templates and Extensions (`tpl_customer_ops`)**, the FDE or business owner configures the exact CRM rules, policies, and tone their industry demands.

# 3. Implementation in `@naelao/runtime`: The Virtual Skill Binary

Because this operates against the in-memory SQLite VFS in `SpaceDO`, the parser runs in **sub-2 milliseconds** using a fast regex/AST parser:

```typescript
// packages/runtime/src/bin/skill.ts
import { VirtualCliCommand } from "@naelao/runtime";

export const SkillCliCommand: VirtualCliCommand = {
  name: "skill",
  description: "Inspects, searches, and reads domain and orchestration skills with section-level targeting.",
  
  execute: async (args, ctx) => {
    const subCommand = args[0]; // "read", "search", "list"
    const skillId = args[1];

    if (subCommand === "list") {
      const skills = await ctx.storage.sqlite.execute("SELECT id, name, description FROM skills");
      return { exitCode: 0, stdout: formatSkillList(skills) };
    }

    if (subCommand === "read") {
      if (!skillId) return { exitCode: 1, stderr: "Usage: skill read <skill_id> [--toc | --section <name>]" };

      // 1. Fetch raw markdown from SQLite VFS in <1ms
      const skillRow = await ctx.storage.sqlite.executeOne<{ content: string }>(
        "SELECT content FROM skills WHERE id = ?", [skillId]
      );
      if (!skillRow) return { exitCode: 1, stderr: `Skill not found: ${skillId}` };

      const markdown = skillRow.content;

      // 2. FLAG: --toc or --headers (Extract headings only)
      if (args.includes("--toc") || args.includes("--headers")) {
        const headers = extractMarkdownHeadings(markdown);
        return { exitCode: 0, stdout: headers };
      }

      // 3. FLAG: --section <name> (Extract specific section)
      const sectionFlagIndex = args.indexOf("--section");
      if (sectionFlagIndex !== -1 && args[sectionFlagIndex + 1]) {
        const targetSection = args[sectionFlagIndex + 1];
        const sectionContent = extractSectionByHeading(markdown, targetSection);
        if (!sectionContent) {
          return { exitCode: 1, stderr: `Section matching "${targetSection}" not found in skill ${skillId}. Use --toc to view sections.` };
        }
        return { exitCode: 0, stdout: sectionContent };
      }

      // 4. Default: Return full markdown (Fallback)
      return { exitCode: 0, stdout: markdown };
    }

    return { exitCode: 1, stderr: "Unknown subcommand. Usage: skill [read|search|list]" };
  }
};

/**
 * Fast AST Helper: Extracts all #, ##, ### headers with section numbers in ~0.5ms
 */
function extractMarkdownHeadings(md: string): string {
  const headingRegex = /^(#{1,3})\s+(.*)$/gm;
  let match;
  const lines = [];
  let index = 1;

  while ((match = headingRegex.exec(md)) !== null) {
    const depth = match[1].length;
    const indent = "  ".repeat(depth - 1);
    lines.push(`${indent}${index}. ${match[0]}`);
    index++;
  }

  return `TABLE OF CONTENTS (Use --section "<Title>" to read targeted section):\n` + lines.join("\n");
}

/**
 * Fast AST Helper: Extracts text from target heading until next equal-or-higher heading
 */
function extractSectionByHeading(md: string, targetHeading: string): string | null {
  const normalizedTarget = targetHeading.toLowerCase();
  const lines = md.split("\n");
  let capturing = false;
  let targetDepth = 0;
  const capturedLines: string[] = [];

  for (const line of lines) {
    const match = line.match(/^(#{1,3})\s+(.*)$/);
    if (match) {
      const depth = match[1].length;
      const title = match[2].toLowerCase();

      if (capturing) {
        // Stop capturing if we hit a heading of equal or higher importance
        if (depth <= targetDepth) {
          break;
        }
      }

      if (title.includes(normalizedTarget)) {
        capturing = true;
        targetDepth = depth;
        capturedLines.push(line);
        continue;
      }
    }

    if (capturing) {
      capturedLines.push(line);
    }
  }

  return capturedLines.length ? capturedLines.join("\n").trim() : null;
}
```

# 6. Implementation in `@naelao/sdk`: The Locale Router

```typescript
// packages/runtime/src/locale/LocaleRouter.ts
import { NaelaoContext } from "@naelao/sdk";

export class LocaleRouter {
  async processInbound(payload: { text?: string; audioBuffer?: ArrayBuffer }, ctx: NaelaoContext) {
    const whitelist = ctx.project.manifest.locale_governance.allowed_locales;

    // 1. CLEF-OMNI CHECKS LANGUAGE IN 38ms (Restricted to whitelist) [2]
    const triage = await ctx.clef.decideFast(payload.text || payload.audioBuffer, {
      detectedLanguage: {
        type: "choice",
        instructions: "Identify which whitelisted language this input belongs to.",
        criteria: whitelist.reduce((acc, l) => {
          acc[l.code] = l.name;
          return acc;
        }, { "UNKNOWN_OOD": "Language outside whitelist or unintelligible noise" })
      },
      confidence: { type: "score", scale: ["low", "high"] }
    });

    const langCode = triage.answers.detectedLanguage.choice;
    const confidence = triage.answers.confidence.score;

    // 2. STOP HALLUCINATIONS: If unconfident or outside whitelist
    if (langCode === "UNKNOWN_OOD" || confidence < 0.70) {
      return {
        haltExecution: true,
        userClarificationPrompt: "I could not verify your language. Please reply in English, Shona, or Ndebele."
      };
    }

    // 3. JIT TRANSLATION TO ENGLISH (If non-English)
    let internalEnglishGoal = payload.text;
    if (langCode !== "en-ZW" && langCode !== "en") {
      internalEnglishGoal = await ctx.tools.translateToEnglish(payload.text || payload.audioBuffer, langCode);
    }

    return {
      haltExecution: false,
      userOriginalLanguage: langCode,
      normalizedEnglishGoal: internalEnglishGoal
    };
  }
}
```

```typescript
// packages/runtime/src/tools/DialectResolverTool.ts
import { registerTool, ToolContext } from "@naelao/sdk";

export const DialectResolverTool = registerTool({
  name: "dialect_resolver",
  description: "Resolves regional code-switching, local slang, and mixed-language phrases into clean operational intent.",
  parameters: {
    phrase: { type: "string", description: "The dialect or slang phrase to resolve" },
    localeHint: { type: "string", description: "e.g. 'sn-ZW', 'hi-IN', 'sw-KE'", optional: true }
  },

  execute: async (ctx: ToolContext, { phrase, localeHint }) => {
    const words = phrase.toLowerCase().split(/\s+/);
    const resolvedTokens: string[] = [];
    let unknownTokens: string[] = [];

    // 1. FAST PATH: Check local SQLite Slang Dictionary in <1ms
    for (const word of words) {
      const match = await ctx.storage.sqlite.executeOne<{ normalized_english: string }>(
        "SELECT normalized_english FROM local_dialect_dictionary WHERE slang_term = ?",
        [word.replace(/[^a-z0-9]/g, "")]
      );

      if (match) {
        resolvedTokens.push(match.normalized_english);
      } else {
        resolvedTokens.push(word);
        unknownTokens.push(word);
      }
    }

    // 2. SLOW PATH: If there are complex grammatical conjugations not in SQLite
    if (unknownTokens.length > 2) {
      // Query lightweight OpenRouter translator or local Laya model
      const translated = await ctx.ai.text.generate({
        profile: "FAST_CODER",
        prompt: `Translate this code-switched dialect phrase into plain English intent: "${phrase}"`
      });
      return { resolvedMeaning: translated.text, source: "NEURAL_TRANSLATION" };
    }

    return { resolvedMeaning: resolvedTokens.join(" "), source: "SQLITE_DICTIONARY_FAST" };
  }
});
```

```typescript
import { NaelaoContext } from "@naelao/sdk";

export class ArtifactsGitMesh {
  constructor(private env: { ARTIFACTS: any; R2: any }) {}

  /**
   * Provisions a dedicated repository for a new Project or Subspace
   */
  async createProjectRepo(projectId: string, jurisdiction: "eu" | "us" = "eu") {
    // Uses Cloudflare's native Artifacts API with data localization
    const { remote, token, repo } = await this.env.ARTIFACTS.create({
      namespace: `naelao_${jurisdiction}`,
      name: `project_${projectId}`
    });

    return { remoteUrl: remote, repoId: repo, writeToken: token };
  }

  /**
   * Commits changes with Naelao's Reasoning Provenance Metadata
   */
  async commitWithReasoning(params: {
    repoId: string;
    branch: string;
    files: Array<{ path: string; content: Uint8Array | string }>;
    commitMessage: string;
    taskMetadata: {
      taskId: string;
      modelId: string;
      reasoningHash: string;
    };
  }) {
    const finalFiles = [];

    for (const file of params.files) {
      const byteLength = typeof file.content === "string" 
        ? new TextEncoder().encode(file.content).length 
        : file.content.byteLength;

      // WORKAROUND: Bypasses Cloudflare Artifacts' 32MB file limit!
      if (byteLength >= 32 * 1024 * 1024) {
        const r2Key = `blobs/${params.repoId}/${crypto.randomUUID()}`;
        await this.env.R2.put(r2Key, file.content);

        // Store lightweight pointer in Git tree
        finalFiles.push({
          path: `${file.path}.naelao-pointer`,
          content: JSON.stringify({
            storage: "R2_POINTER",
            r2_key: r2Key,
            size_bytes: byteLength,
            sha256: params.taskMetadata.reasoningHash
          })
        });
      } else {
        finalFiles.push(file);
      }
    }

    // Extended Git commit message injecting cognitive trailers
    const cognitiveCommitMessage = `
${params.commitMessage}

X-Naelao-Task: ${params.taskMetadata.taskId}
X-Naelao-Model: ${params.taskMetadata.modelId}
X-Naelao-Reasoning: ${params.taskMetadata.reasoningHash}
`.trim();

    // Commit directly via Cloudflare Artifacts Workers binding
    return await this.env.ARTIFACTS.commit({
      repo: params.repoId,
      branch: params.branch,
      files: finalFiles,
      message: cognitiveCommitMessage
    });
  }
```

```typescript
    // e.g. "agent+space123+task456@inbox.avos.run"

    const parser = new PostalMime();

    for (const file of parsedEmail.attachments) {
      const r2Key = `inbound_mail/${spaceId}/${crypto.randomUUID()}`;
      await env.R2.put(r2Key, file.content);

      sender: message.from,
```
