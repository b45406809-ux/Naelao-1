```typescript
// workers/billing/src/GeoPaymentRouter.ts
export class GeoPaymentRouter {
  
  /**
   * Returns the optimal checkout session based on user location
   */
  async createCheckoutSession(req: Request, params: {
    spaceId: string;
    planId: "CONSUMER_7" | "SMB_29" | "STUDIO_199";
    amountUSD: number;
    userEmail: string;
  }, env: Env) {
    
    // 1. Detect visitor country code directly from Cloudflare Anycast header
    const countryCode = req.headers.get("CF-IPCountry") || "US";

    console.log(`🌍 Inbound checkout detected from country: [${countryCode}]`);

    // -------------------------------------------------------------------------
    // ROUTE A: ZIMBABWE (Paynow Gateway)
    // -------------------------------------------------------------------------
    if (countryCode === "ZW") {
      return await this.createPaynowCheckout({
        reference: `sub_${params.spaceId}_${Date.now()}`,
        amountUSD: params.amountUSD,
        userEmail: params.userEmail,
        returnUrl: `https://app.naelao.com/workspace?status=success`
      }, env);
    }

    // -------------------------------------------------------------------------
    // ROUTE B: INDIA (Razorpay / UPI Gateway)
    // -------------------------------------------------------------------------
    if (countryCode === "IN") {
      return await this.createRazorpayCheckout(params, env);
    }

    // -------------------------------------------------------------------------
    // ROUTE C: DEFAULT WORLDWIDE (Lemon Squeezy / Stripe via UK Entity)
    // -------------------------------------------------------------------------
    return await this.createLemonSqueezyCheckout({
      variantId: this.getLemonVariantId(params.planId),
      customData: { spaceId: params.spaceId },
      userEmail: params.userEmail
    }, env);
  }

  private async createPaynowCheckout(data: any, env: Env) {
    // Calls Paynow API directly with your personal Paynow Integration ID & Key
    const payload = new URLSearchParams({
      id: env.PAYNOW_INTEGRATION_ID,
      reference: data.reference,
      amount: data.amountUSD.toFixed(2),
      additionalinfo: "Naelao AI Operating System Plan",
      returnurl: data.returnUrl,
      resulturl: "https://api.naelao.com/api/webhooks/paynow",
      authemail: data.userEmail,
      status: "Message"
    });

    // Hash signature calculation (SHA-512)
    payload.append("hash", this.generatePaynowHash(payload, env.PAYNOW_INTEGRATION_KEY));

    const res = await fetch("https://www.paynow.co.zw/interface/initiatetransaction", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: payload.toString()
    });

    const parsed = new URLSearchParams(await res.text());
    return {
      gateway: "PAYNOW_ZW",
      redirectUrl: parsed.get("browserurl"), // User redirected to Paynow/EcoCash screen
      pollUrl: parsed.get("pollurl")
    };
  }

  private async createLemonSqueezyCheckout(data: any, env: Env) {
    // Calls Lemon Squeezy API directly
    const res = await fetch("https://api.lemonsqueezy.com/v1/checkouts", {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${env.LEMON_SQUEEZY_API_KEY}`,
        "Content-Type": "application/vnd.api+json"
      },
      body: JSON.stringify({
        data: {
          type: "checkouts",
          attributes: {
            checkout_data: { custom: data.customData, email: data.userEmail }
          },
          relationships: {
            store: { data: { type: "stores", id: env.LEMON_SQUEEZY_STORE_ID } },
            variant: { data: { type: "variants", id: data.variantId } }
          }
        }
      })
    });

    const json = await res.json();
    return {
      gateway: "LEMON_SQUEEZY_GLOBAL",
      redirectUrl: json.data.attributes.url // International users get Apple Pay / Card
    };
  }
}
```

### The Reality of India’s Payment System

India has one of the most advanced digital payment ecosystems in the world, dominated by **UPI (Unified Payments Interface)**—over 80% of digital transactions in India happen through apps like Google Pay, PhonePe, and Paytm.

However, the **Reserve Bank of India (RBI)** enforces strict financial regulations [3, 4]:
1. **You CANNOT directly register on domestic Indian gateways** (like domestic Razorpay, Cashfree, or Paytm) as an individual in Zimbabwe. They legally require an Indian **PAN Card** (tax number), **GST registration**, and an **Indian local bank account** [1, 2].
2. **The "Recurring E-Mandate" Barrier:** In India, automatic monthly credit card recurring billing (e.g. standard $29/mo auto-debit) is heavily restricted [3, 4]. The RBI legally mandates an SMS One-Time Password (OTP) or an Additional Factor of Authentication (AFA) before any recurring card deduction can occur [3, 4].

The viable way for a non-Indian resident to accept payments from India—**without setting up an Indian company or providing proof of residency in India**—is to use a cross-border Merchant of Record.

```typescript
// workers/billing/src/GeoPaymentRouter.ts

export class GeoPaymentRouter {
  async getCheckoutUrl(req: Request, params: { spaceId: string; planId: string; amountUSD: number }) {
    const country = req.headers.get("CF-IPCountry") || "US";

    // 1. ZIMBABWE ──► Paynow (EcoCash / InnBucks) - No foreign company needed
    if (country === "ZW") {
      return await this.createPaynowSession(params);
    }

    // 2. INDIA ──► Dodo Payments (UPI QR Code + RuPay) - Uses your UK LTD, Zero Indian Docs
    if (country === "IN") {
      const inrEquivalent = Math.round(params.amountUSD * 84); // Approx conversion to INR
      
      const res = await fetch("https://api.dodopayments.com/v1/checkouts", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${env.DODO_PAYMENTS_API_KEY}`,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          payment_methods: ["upi", "card"],
          total_amount: inrEquivalent,
          currency: "INR",
          metadata: { spaceId: params.spaceId, type: "GAS_TANK_TOPUP" }
        })
      });
      const data = await res.json();
      return data.checkout_url;
    }

    // 3. REST OF WORLD ──► Lemon Squeezy / Paddle (Apple Pay / Cards) - Uses your UK LTD
    return await this.createLemonSqueezySession(params);
  }
}
```

```typescript
// packages/runtime/src/sentry/AdaptiveDecisionClient.ts
import { NaelaoContext } from "@naelao/sdk";

export class AdaptiveDecisionClient {
  
  async decide(params: {
    state: string;
    questions: Record<string, any>;
    priority: "INTERACTIVE_SPEED" | "BACKGROUND_BULK" | "HIGH_ACCURACY";
  }, ctx: NaelaoContext) {

    // 1. INTERACTIVE SPEED PRIORITY (38.8ms via Native Workers AI) [2]
    if (params.priority === "INTERACTIVE_SPEED") {
      try {
        return await ctx.env.AI.run("@cf/cloudflare/clef-flash", {
          state: params.state,
          questions: params.questions
        });
      } catch (err) {
        console.warn("Workers AI Clef-flash busy; falling over to Decider Flash V1.2 on OpenRouter...", err);
      }
    }

    // 2. BACKGROUND BULK PRIORITY (The 1-Cent King via OpenRouter)
    if (params.priority === "BACKGROUND_BULK") {
      return await ctx.vault.fetchOpenRouter({
        model: "perplexity/decider-flash-v1.2", // $0.01 / M tokens!
        messages: [{
          role: "user",
          content: JSON.stringify({ state: params.state, questions: params.questions })
        }]
      });
    }

    // 3. HIGH ACCURACY COMPLEX TRIAGE ($0.02 / M)
    return await ctx.vault.fetchOpenRouter({
      model: "perplexity/decider-v1.2",
      messages: [{
        role: "user",
        content: JSON.stringify({ state: params.state, questions: params.questions })
      }]
    });
  }
}
```

* Cloudflare does not run all your projects on a single CPU core in a local server rack.
* Cloudflare automatically runs Project A in London, Project B in Johannesburg, and Project C in Singapore—closest to the external webhooks and services they interact with.
* **The compute load is physically distributed across thousands of edge servers.**

### 4. Fault Isolation (Zero Blast Radius)
* If an experimental Python script crashes in Project C, **Project A and Project B continue running in the background without dropping a single frame.**

### Part 4: How Naelao Makes Worker-to-Worker Chatting Safe

To prevent the UC Berkeley 86% failure rate (infinite conversational loops and token drain), Naelao replaces the blunt "No-Delegate" rule with **Three Universal Invariants**:

#### Invariant 1: The Attenuated Peer Channel (No Spawning)
Leaf workers are allowed to communicate with peer workers, but **they are given an explicit Peer Grant**:
```typescript
interface PeerChannelGrant {
  recipientWorkerId: string;    // Who can I talk to?
  maxTurnBudget: number;         // Hard limit (e.g. max 3 exchanges)
  canSpawnNewWorkers: false;     // 🛑 STILL BANNED from spawning new agents!
}
```
* Worker A can ask Worker B: *"What is the export name of your auth function?"*
* Worker B can reply: *`export function verifyAuth()`*.
* **Neither worker can spawn a new child agent.** The population of the swarm remains fixed.

#### Invariant 2: The Shared Turn Gas Tank
Instead of giving each worker an infinite loop:
* The pair of debating or collaborating workers share a **Micro-Gas Envelope (e.g., $0.01 max)**.
* When the turn counter hits 3 or the budget reaches $0.01, **the communication channel automatically severs.**
* If they have not reached consensus, the conversation is dumped to the Lead Planner to make the executive tie-breaker decision.

#### Invariant 3: Clef-Flash as the Turn Arbiter (38ms Sentry)
Between every peer-to-peer exchange, **Clef-flash** evaluates the message in **38ms** [2]:
* Is this message an unproductive loop (*"I agree," "No, I agree"*)?
* Has agreement been reached?
* If agreement is detected, Clef halts the debate early, saving tokens.

```typescript
    let currentTurns = 0;

    // 1. Initialize mailboxes for all participants
    for (const actorId of config.participants) {
      messageBus.set(actorId, []);
    }

    // 2. The Universal Dispatch Loop
    while (currentTurns < 20) {
      currentTurns++;

      // Evaluate active workers based on the topology's routing policy
      for (const actorId of config.participants) {
        const inbox = messageBus.get(actorId) || [];
        
        // Only run worker if it has messages, or is the root trigger
        if (inbox.length === 0 && currentTurns > 1) continue;

        // Execute the worker turn
        const workerResponse = await ctx.workers.execute(actorId, {
          inboxMessages: inbox,
          goal: goal,
          blackboard: config.sharedBlackboardTable 
            ? await ctx.storage.sqlite.execute(`SELECT * FROM ${config.sharedBlackboardTable}`)
            : null
        });

        // Clear processed inbox
        messageBus.set(actorId, []);

        // 3. ROUTING POLICY CHECK: Who is this worker allowed to message?
        if (workerResponse.outboundHandoff) {
          const { targetActorId, payload } = workerResponse.outboundHandoff;
          
          const rule = config.peerRoutingRules.find(r => r.from === actorId && r.to === targetActorId);
          if (!rule) {
            throw new Error(`Topology Violation: ${actorId} is not permitted to message ${targetActorId}`);
          }

          if (currentTurns > rule.maxTurns) {
            console.warn(`[Swarm] Turn budget reached between ${actorId} and ${targetActorId}. Halting.`);
            break;
          }

          // Deliver message directly to peer's inbox!
          messageBus.get(targetActorId)?.push({ from: actorId, text: payload });
        }

        // 4. CHECK TERMINATION: Has the goal been committed to SQLite / Git?
        if (workerResponse.isGoalCompleted) {
          return { status: "COMPLETED", turns: currentTurns };
        }
      }
    }

    return { status: "MAX_TURNS_REACHED", turns: currentTurns };
  }
}
```

# 4. Implementation in `@naelao/sdk`: The Swarm Governance Engine

Here is how the Hypervisor enforces tier limits and configures the UI automatically:

```typescript
// packages/runtime/src/governance/SwarmGovernanceEngine.ts
import { NaelaoContext } from "@naelao/sdk";

export interface PlanLimits {
  maxWorkers: number;
  maxDepth: number;
  allowedTopologies: string[];
}

export const PLAN_LIMITS: Record<string, PlanLimits> = {
  CONSUMER: {
    maxWorkers: 1,
    maxDepth: 1,
    allowedTopologies: ["DIRECT_WORKER", "PIPELINE_RELAY"]
  },
  SMB_PRO: {
    maxWorkers: 3,
    maxDepth: 2,
    allowedTopologies: ["DIRECT_WORKER", "PIPELINE_RELAY", "SPECULATIVE_RACE", "DIALECTICAL_DEBATE"]
  },
  ENTERPRISE: {
    maxWorkers: 12,
    maxDepth: 3,
    allowedTopologies: ["ALL"]
  }
};

export class SwarmGovernanceEngine {
  /**
   * Evaluates if the requested swarm topology is legal under user's plan.
   * Clamps automatically if plan limits are exceeded!
   */
  clampSwarmToPlan(requestedTopology: string, userPlanTier: string): {
    approvedTopology: string;
    maxConcurrency: number;
    maxDepth: number;
  } {
    const limits = PLAN_LIMITS[userPlanTier] || PLAN_LIMITS.CONSUMER;

    // Check if topology is unlocked on this tier
    const isAllowed = limits.allowedTopologies.includes("ALL") || 
                      limits.allowedTopologies.includes(requestedTopology);

    if (!isAllowed) {
      console.warn(`[Governance] Plan ${userPlanTier} does not permit ${requestedTopology}. Clamping to DIRECT_WORKER.`);
      return {
        approvedTopology: "DIRECT_WORKER",
        maxConcurrency: 1,
        maxDepth: 1
      };
    }

    return {
      approvedTopology: requestedTopology,
      maxConcurrency: limits.maxWorkers,
      maxDepth: limits.maxDepth
    };
  }
}
```

### The Strategic Summary

1. **Naelao is for everyone:** The consumer gets an **unbreakable, private Chief of Staff** that runs their personal life over WhatsApp for $6.99/month.
2. **Invisible Complexity:** Consumers don't see Git, Docker, or swarms; they just experience a helpful, memory-backed assistant that understands their local context.
3. **High Unit Economics:** Because personal tasks run on hyper-efficient edge models (Clef and DeepSeek V4) [2, 4], consumer accounts cost **less than $0.40/month to host**, yielding a **95%+ gross profit margin**.
4. **The Bottom-Up Flywheel:** The person using Naelao for personal budgeting today is the entrepreneur launching an autonomous venture or selling a vertical template tomorrow.
