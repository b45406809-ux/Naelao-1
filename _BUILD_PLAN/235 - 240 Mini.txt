### 1. The Vectorize Metadata ACL Schema
In `packages/rag` and `workers/knowledge/src/index.ts`, every document chunk indexed into **Cloudflare Vectorize** is stamped with integer clearance ranks:
```typescript
// Ranked roles: Viewer (1) -> Member (2) -> Editor (3) -> Admin (4) -> Owner (5)
export const ROLE_HIERARCHY_RANK: Record<string, number> = {
  viewer: 1,
  member: 2,
  editor: 3,
  admin: 4,
  owner: 5
};
export interface ChunkVectorMetadata {
  space_id: string;
  project_id: string; // "" = Space-wide, otherwise specific projectId
  min_role_rank: number; // e.g. 4 for Admin-only documents
  doc_id: string;
  title: string;
  text: string;
}
```

### 2. The Hardware-Level ACL Query (Zero Data Leakage)
When a user asks a question, Vectorize filters out unauthorized documents **inside the vector index before computing similarity scores**:
```typescript
// workers/knowledge/src/index.ts
export async function queryPermissionedKnowledge(
  vectorize: VectorizeIndex,
  queryVector: number[],
  spaceId: string,
  userRole: string,
  activeProjectId?: string
) {
  const userRank = ROLE_HIERARCHY_RANK[userRole] ?? 1;
  // Multi-tier ACL filter:
  // 1. Must belong to this space
  // 2. Either Space-Wide (project_id == "") OR this specific active project
  // 3. User's rank must be >= document's minimum required rank
  const matches = await vectorize.query(queryVector, {
    topK: 6,
    filter: {
      space_id: spaceId,
      project_id: { $in: [activeProjectId || "", ""] },
      min_role_rank: { $lte: userRank }
    },
    returnMetadata: "all"
  });
  return matches.matches.map(m => m.metadata as ChunkVectorMetadata);
}
```

```typescript
// packages/rag/src/hybrid.ts
export async function googleModeHybridSearch(
  query: string,
  env: Env,
  acl: { spaceId: string; userRole: string; projectId?: string }
) {
  // 1. Check Semantic Cache (Sub-50ms)
  const queryEmbedding = await env.AI.run("@cf/baai/bge-small-en-v1.5", { text: [query] });
  const cacheHit = await env.SEMANTIC_CACHE_VECTORS.query(queryEmbedding.data[0], {
    topK: 1,
    filter: { space_id: acl.spaceId }
  });
  if (cacheHit.matches.length > 0 && cacheHit.matches[0].score > 0.96) {
    const cached = await env.SEMANTIC_CACHE_KV.get(cacheHit.matches[0].id);
    if (cached) return { text: cached, source: "SEMANTIC_CACHE", latencyMs: 35 };
  }
  // 2. Parallel Dense (Vectorize) + Sparse (BM25)
  const denseResults = await queryPermissionedKnowledge(
    env.KB_VECTORS, queryEmbedding.data[0], acl.spaceId, acl.userRole, acl.projectId
  );
  // Run BM25 keyword matching across candidate document chunks
  const sparseResults = bm25Score(query, denseResults.map(d => ({ id: d.doc_id, content: d.text })));
  // 3. Reciprocal Rank Fusion (RRF)
  const fusedChunks = reciprocalRankFusion(
    denseResults.map((d, idx) => ({ id: d.doc_id, score: 1 / (idx + 1) })),
    sparseResults
  );
  return {
    chunks: fusedChunks,
    citations: denseResults.map((d, i) => ({
      index: i + 1,
      title: d.title,
      docId: d.doc_id,
      snippet: d.text.slice(0, 150)
    }))
  };
}
```

### 2. The Branching & Pull Request Engine Implementation
```typescript
// workers/agents/src/script-engine/prebuilt/workspace-branch.ts
export class WorkspaceBranchEngine {
  constructor(private doStorage: DurableObjectStorage) {}
  // 1. Fork a new virtual branch from 'main' (Copy-on-Write snapshot in SQLite)
  async createBranch(sourceBranch = "main", newBranch: string) {
    this.doStorage.sql.exec(`
      INSERT OR REPLACE INTO workspace_files (branch, path, content, hash, size_bytes, updated_at)
      SELECT ?, path, content, hash, size_bytes, datetime('now')
      FROM workspace_files WHERE branch = ?
    `, newBranch, sourceBranch);
    return { ok: true, branch: newBranch };
  }
  // 2. Generate a Pull Request with AI-generated diff summary
  async createPullRequest(params: {
    spaceId: string;
    projectId: string;
    sourceBranch: string;
    title: string;
    reasoningSummary: string;
    authorUserId: string;
  }) {
    const prId = `pr_${Date.now()}`;
    // Compute diffs between sourceBranch and main
    const sourceFiles = this.doStorage.sql.exec(`SELECT path, content, hash FROM workspace_files WHERE branch = ?`, params.sourceBranch).toArray() as any[];
    const mainFiles = this.doStorage.sql.exec(`SELECT path, content, hash FROM workspace_files WHERE branch = 'main'`).toArray() as any[];
    const mainMap = new Map(mainFiles.map(f => [f.path, f]));
    const diffManifest: Array<{ path: string; status: "MODIFIED" | "CREATED" | "DELETED"; oldContent?: string; newContent?: string }> = [];
    for (const src of sourceFiles) {
      const main = mainMap.get(src.path);
      if (!main) {
        diffManifest.push({ path: src.path, status: "CREATED", newContent: src.content });
      } else if (main.hash !== src.hash) {
        diffManifest.push({ path: src.path, status: "MODIFIED", oldContent: main.content, newContent: src.content });
      }
    }
    this.doStorage.sql.exec(`
      INSERT INTO pull_requests (id, space_id, project_id, source_branch, target_branch, title, reasoning_summary, diff_manifest_json, author_user_id)
      VALUES (?, ?, ?, ?, 'main', ?, ?, ?, ?)
    `, prId, params.spaceId, params.projectId, params.sourceBranch, params.title, params.reasoningSummary, JSON.stringify(diffManifest), params.authorUserId);
    return { ok: true, prId, changedFiles: diffManifest.length, manifest: diffManifest };
  }
  // 3. Atomic Merge to Main Workspace
  async mergePullRequest(prId: string) {
    const pr = this.doStorage.sql.exec(`SELECT * FROM pull_requests WHERE id = ? AND status = 'OPEN'`, prId).toArray()[0] as any;
    if (!pr) throw new Error("PR not found or already merged.");
    // Atomically copy all modified files from sourceBranch into 'main'
    this.doStorage.transactionSync(() => {
      this.doStorage.sql.exec(`
        INSERT OR REPLACE INTO workspace_files (branch, path, content, hash, size_bytes, updated_at)
        SELECT 'main', path, content, hash, size_bytes, datetime('now')
        FROM workspace_files WHERE branch = ?
      `, pr.source_branch);
      this.doStorage.sql.exec(`
        UPDATE pull_requests SET status = 'MERGED', merged_at = datetime('now') WHERE id = ?
      `, prId);
    });
    return { ok: true, message: `✔ Successfully merged PR #${prId} into Main Workspace.` };
  }
}
```

```typescript
// workers/agents/src/services/ArtifactsGitService.ts
export class ArtifactsGitService {
  constructor(private env: Env) {}
  // 1. Create a fresh Git Repository for a new Project
  async createProjectRepo(projectId: string) {
    const repoName = `proj_${projectId}`;
    const repo = await this.env.AGENT_REPOS.create(repoName);
    const token = await repo.issueToken({ permission: "write", ttlSeconds: 86400 });
    
    return {
      repoId: repo.id,
      remoteUrl: repo.remote, // https://x:token@artifacts.cloudflare.net/...
      token: token.token
    };
  }
  // 2. Instant Zero-Copy Fork for Tier 4 Private Scratchpad
  async forkForPrivateScratchpad(parentRepoName: string, scratchpadId: string) {
    const forkName = `${parentRepoName}_fork_${scratchpadId}`;
    const fork = await this.env.AGENT_REPOS.fork({
      source: parentRepoName,
      name: forkName
    });
    const writeToken = await fork.issueToken({ permission: "write", ttlSeconds: 3600 });
    return {
      forkName,
      remoteUrl: fork.remote,
      token: writeToken.token
    };
  }
  // 3. Generate Native Git Diff for Pull Request Review Card
  async generatePullRequestDiff(baseRepoName: string, forkRepoName: string) {
    const baseRepo = await this.env.AGENT_REPOS.get(baseRepoName);
    const forkRepo = await this.env.AGENT_REPOS.get(forkRepoName);
    // Fetch commit comparison between fork and base
    const diff = await this.env.AGENT_REPOS.compare({
      base: baseRepo.head,
      head: forkRepo.head
    });
    return {
      baseCommit: baseRepo.head,
      headCommit: forkRepo.head,
      changedFiles: diff.files, // [{ path, additions, deletions, patch }]
      totalAdditions: diff.additions,
      totalDeletions: diff.deletions
    };
  }
  // 4. Atomic Fast-Forward Merge to Main Workspace
  async mergePullRequest(baseRepoName: string, forkRepoName: string) {
    const baseRepo = await this.env.AGENT_REPOS.get(baseRepoName);
    const forkRepo = await this.env.AGENT_REPOS.get(forkRepoName);
    // Fast-forward base repository to the fork's commit head
    await baseRepo.fastForward(forkRepo.head);
    return {
      ok: true,
      mergedCommit: forkRepo.head,
      message: `✔ Successfully merged ${forkRepoName} into ${baseRepoName} at commit ${forkRepo.head.slice(0, 7)}.`
    };
  }
}
```
