### 3. Implementation: Routing Prompts to the Local Tunnel
In `workers/agents/src/shared/ai-client.ts`, when a model starts with `local/` or `ollama/`, route the request through `LocalTunnelDO`:
```typescript
// workers/agents/src/shared/ai-client.ts
export async function dispatchLocalModelInference(
  modelName: string,
  requestBody: any,
  env: Env,
  ctx: { spaceId: string; userId: string }
) {
  // 1. Get the LocalTunnelDO stub for this space and user
  const tunnelDOId = env.LOCAL_TUNNEL.idFromName(`${ctx.spaceId}:${ctx.userId}`);
  const tunnelDO = env.LOCAL_TUNNEL.get(tunnelDOId);
  // 2. Format as standard OpenAI-compatible completions for Ollama
  const cleanModel = modelName.replace(/^(local\/|ollama\/)/, "");
  const payload = {
    ...requestBody,
    model: cleanModel
  };
  // 3. Dispatch through the WebSocket tunnel to the on-premise GPU
  const response = await tunnelDO.fetch("http://internal/exec", {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({
      op: "ollama_chat",
      params: payload
    })
  });
  return await response.json();
}
```

### 4. The Local Bridge Client (`packages/local-agent/index.mjs`)
The small Node.js CLI daemon running on the company's GPU machine:
```javascript
// packages/local-agent/index.mjs
import WebSocket from "ws";
export async function startLocalOllamaBridge({ url, spaceId, token, ollamaUrl = "http://localhost:11434" }) {
  const ws = new WebSocket(`${url}/local-tunnel/connect?space=${spaceId}&token=${token}`);
  ws.on("open", () => console.log("✔ Connected to AVOS Edge Tunnel. Local Ollama is active."));
  ws.on("message", async (data) => {
    const request = JSON.parse(data.toString());
    if (request.op === "ollama_chat") {
      try {
        // Forward prompt to local Ollama instance on localhost:11434
        const ollamaRes = await fetch(`${ollamaUrl}/v1/chat/completions`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(request.params)
        });
        const result = await ollamaRes.json();
        ws.send(JSON.stringify({ request_id: request.request_id, ok: true, result }));
      } catch (err) {
        ws.send(JSON.stringify({ request_id: request.request_id, ok: false, error: err.message }));
      }
    }
  });
}
```

### 5. Enterprise Security: The "Air-Gapped Space" Lock
In **Space Settings**, enterprise admins can enable **`Air-Gapped Mode`**:
```typescript
export interface EnterpriseCompliancePolicy {
  airGappedOnly: boolean; // When true: ALL cloud AI APIs (Anthropic/OpenAI/Workers AI) are blocked
  allowedLocalTunnels: string[]; // Whitelisted on-prem GPU machine IDs
}
```
* **The Guarantee:** If `airGappedOnly === true`, the router and `NaelaoAgentDO` **physically refuse** to dispatch calls to any external cloud URL. If the local Ollama machine goes offline, the system halts with a security alert rather than silently falling back to a cloud model.
